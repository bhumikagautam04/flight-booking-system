CREATE TABLE flight (
    flight_id INT AUTO_INCREMENT PRIMARY KEY,
    flight_name VARCHAR(45),
    flight_type VARCHAR(45),
    flight_seat INT,
    days INT,
    
    source_city VARCHAR(45),
    departure_time TIME,

    destination_city varchar(45),
    arrival_time TIME,

    company VARCHAR(45),
    logo VARCHAR(255),

    FOREIGN KEY (source_city)
    REFERENCES city(city_id),

    FOREIGN KEY (destination_city)
    REFERENCES city(city_id)
);
